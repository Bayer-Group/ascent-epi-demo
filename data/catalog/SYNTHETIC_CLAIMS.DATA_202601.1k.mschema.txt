# Table: dx
[
(enc_id:VARCHAR, encounter this record belongs to; joins ENCOUNTER.ENC_ID),
(mrn:VARCHAR, site-local registration id, NOT a patient key -- one person may hold several; join PATIENT_MASTER and count MEMBER_ID to count people, Examples: [LSR0000875, LSR0000476, LSR0000405, LSR0000209, LSR0000219, LSR0000664]),
(dx_cd:VARCHAR, diagnosis code, written WITH the decimal point throughout this table, such as I10.9, Examples: [Q40.0, R01.0, Q30.0, R78.0, Q31.0, R72.0]),
(dx_cd_type:VARCHAR, code system of DX_CD: DX10 and D10 are BOTH ICD-10 and DX9 is ICD-9, all three at every facility -- match all of them or you silently drop patients, Examples: [DX10, DX9, D10]),
(dx_desc:VARCHAR, diagnosis description as free text, Examples: [Essential (primary) hypertension, Obesity, unspecified, Pure hypercholesterolemia, Prediabetes, Hypertensive heart disease without heart failure, Hypertension, unspecified]),
(dx_dt:DATE, date the diagnosis was recorded, Examples: [MIN: 2001-01-04, MAX: 2091-10-26]),
(primary_fl:VARCHAR, flag marking the primary diagnosis of the encounter, Examples: [N, Y]),
(dx_seq:BIGINT, position of the diagnosis within the encounter, Examples: [1, 2, 3, 4]),
(fac_cd:VARCHAR, facility that produced the record; joins FACILITY.FAC_CD, Examples: [LSR, NGH, RBM, SVO, PAC]),
]

(admit_dt:DATE, admission date; a handful of rows carry implausible future dates, seeded so the demo has dirty data to handle, Examples: [MIN: 2001-12-31, MAX: 2091-10-11]),
(disch_dt:DATE, discharge date, Examples: [MIN: 2002-10-02, MAX: 2071-07-28]),
(los_days:BIGINT, length of stay in days, Examples: [0, 1, 2, 3, 4, 5]),
(dept_nm:VARCHAR, department name, Examples: [Lakeside Endocrinology Associates, Northgate Cardiology Clinic, Riverbend Family Practice, Summit Valley Oncology Institute, Pinehurst Ambulatory Care, Lakeside Regional Health]),
(attending_id:VARCHAR, attending clinician identifier, Examples: [PRV0131, PRV0141, PRV0115, PRV0123, PRV0128, PRV0111]),
(fac_cd:VARCHAR, facility that produced the record; joins FACILITY.FAC_CD, Examples: [LSR, NGH, RBM, SVO, PAC]),
(total_charge_amt:VARCHAR, total amount charged for the encounter),
]
